using System;
using System.Threading.Tasks;
using Unity.Netcode;
using Unity.Services.Authentication;
using Unity.Services.Core;
using UnityEngine;

public class MultiplayerManager : MonoBehaviour
{
    public static MultiplayerManager Instance { get; private set; }
    public Task InitializationTask => initializationTask;
    private Task initializationTask;

    private async void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        DontDestroyOnLoad(gameObject);

        initializationTask = InitializeAsync();
        await initializationTask;
    }

    private async Task InitializeAsync()
    {
        try
        {
            await UnityServices.InitializeAsync();

            if (!AuthenticationService.Instance.IsSignedIn)
            {
                await AuthenticationService.Instance.SignInAnonymouslyAsync();
                Debug.Log("Unity Services initialized and player signed in");
            } else
                Debug.Log("Unity Services initialized. Player already signed in");
        }
        catch (Exception ex)
        {
            Debug.LogError($"Failed Initializing Unity Services: {ex}");
        }
    }

    private async void Start()
    {
        await initializationTask;

        // TEMP: local testing
        StartHost();
        // or StartClient();
    }

    public void StartHost()
    {
        NetworkManager.Singleton.StartHost();
    }

    public void StartClient()
    {
        NetworkManager.Singleton.StartClient();
    }
}
